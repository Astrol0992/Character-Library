// ── DATA ───────────────────────────────────────────
function genId() { return Date.now().toString(36) + Math.random().toString(36).slice(2,8); }

function loadData() {
  try {
    var s = localStorage.getItem('charlib');
    if (s) {
      var p = JSON.parse(s);
      if ((p.v||0) >= (__D__.v||0)) { chars = p.d||{}; return; }
    }
  } catch(e) {}
  chars = JSON.parse(JSON.stringify(__D__.d||{}));
  save();
}

function save() {
  try { localStorage.setItem('charlib', JSON.stringify({v:Date.now(),d:chars})); }
  catch(e) { console.warn('Storage full',e); }
}

// ── UTILS ──────────────────────────────────────────
function esc(s) {
  return String(s||'').replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;').replace(/"/g,'&quot;');
}
function getFiltered() {
  var lq = q.toLowerCase();
  return Object.values(chars).filter(function(c){return c.name.toLowerCase().indexOf(lq)>-1;})
    .sort(function(a,b){return a.name.localeCompare(b.name);});
}

// ── IMAGE UTILS ────────────────────────────────────
function compressImg(src) {
  return new Promise(function(res) {
    var img = new Image();
    img.onload = function() {
      var r = Math.min(1, 1000/img.width);
      var cv = document.createElement('canvas');
      cv.width = Math.round(img.width*r); cv.height = Math.round(img.height*r);
      cv.getContext('2d').drawImage(img,0,0,cv.width,cv.height);
      res(cv.toDataURL('image/jpeg',0.72));
    };
    img.onerror = function(){res(src);};
    img.src = src;
  });
}
function readFile(file) {
  return new Promise(function(res,rej) {
    var fr = new FileReader();
    fr.onload = function(e){res(e.target.result);}; fr.onerror = rej;
    fr.readAsDataURL(file);
  });
}
async function handleImgFile(file) {
  if (!file||!file.type.startsWith('image/')) return;
  try { var raw = await readFile(file); _mi = await compressImg(raw); _mt='device'; renderModal(); } catch(e){}
}

function copyMD() {
  var c=pg?chars[pg]:null; if(!c) return;
  var md='## '+c.name+'\n\n'+c.images.map(function(i){
    return '### '+i.label+'\n!['+i.label+']('+i.url+')';
  }).join('\n\n');
  var done=function(){
    var btn=document.getElementById('md-btn');
    if(btn){btn.innerHTML='&#10003; Copied!';btn.className=btn.className.replace('bg','bsuc');}
    clearTimeout(_tmr);
    _tmr=setTimeout(function(){
      var b=document.getElementById('md-btn');
      if(b){b.innerHTML='&#8860; Rentry MD';b.className=b.className.replace('bsuc','bg');}
    },2500);
  };
  try {
    navigator.clipboard.writeText(md).then(done).catch(function(){fbCopy(md);done();});
  } catch(e) { fbCopy(md);done(); }
}
function fbCopy(txt) {
  var ta=document.createElement('textarea');
  ta.value=txt;ta.style.cssText='position:fixed;top:0;opacity:0';
  document.body.appendChild(ta);ta.focus();ta.select();
  try{document.execCommand('copy');}catch(e){}
  document.body.removeChild(ta);
}
